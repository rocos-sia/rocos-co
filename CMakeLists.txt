cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ROCOS_CO_VERSION LIMIT_COUNT 1)
if(NOT ROCOS_CO_VERSION MATCHES "^([0-9]+[.][0-9]+[.][0-9]+)-rc[.]([1-9][0-9]*)$")
    message(FATAL_ERROR "Malformed internal version in VERSION")
endif()
project(rocos_co VERSION "${CMAKE_MATCH_1}" LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS VERSION)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "rocos-shm ABI v2 requires Linux x86_64")
endif()
message(STATUS "ROCOS-CO version ${ROCOS_CO_VERSION}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
if(CMAKE_INSTALL_DOCDIR STREQUAL "" OR CMAKE_INSTALL_DOCDIR STREQUAL "share/doc/rocos_co")
    set(CMAKE_INSTALL_DOCDIR "${CMAKE_INSTALL_DATAROOTDIR}/doc/rocos-co")
endif()
include(CTest)
include(cmake/Dependencies.cmake)
configure_file(cmake/version.hpp.in generated/version.hpp @ONLY)
add_subdirectory(src)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
install(DIRECTORY config/ DESTINATION ${CMAKE_INSTALL_DATADIR}/rocos-co/config)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
foreach(dep IN ITEMS CANOPENNODE ROCOS_SHM)
    install(FILES "${${dep}_SOURCE_DIR}/LICENSE" DESTINATION ${CMAKE_INSTALL_DOCDIR}/licenses RENAME "${dep}.txt")
endforeach()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(EXISTS "/usr/share/doc/libyaml-cpp-dev/copyright")
    install(FILES "/usr/share/doc/libyaml-cpp-dev/copyright" DESTINATION ${CMAKE_INSTALL_DOCDIR}/licenses RENAME yaml-cpp.txt)
endif()
